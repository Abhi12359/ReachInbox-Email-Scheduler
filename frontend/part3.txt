                  onChange={(e) => setRecipientInput(e.target.value)}
                  onKeyDown={(e) => {
                    if (
                      e.key === "Enter" ||
                      e.key === "," ||
                      e.key === ";"
                    ) {
                      e.preventDefault();
                      addRecipient();
                    }
                  }}
                />

                <button
                  className="upload-list"
                  onClick={() => fileRef.current?.click()}
                >
                  ↑ Upload List
                </button>

                <input
                  ref={fileRef}
                  type="file"
                  accept=".csv,.txt"
                  hidden
                  onChange={(e) =>
                    uploadList(e.target.files?.[0])
                  }
                />
              </div>
            </div>

            {/* SUBJECT */}

            <div className="compose-line">
              <label>
                Subject
              </label>

              <input
                value={subject}
                placeholder="Subject"
                onChange={(e) => setSubject(e.target.value)}
              />
            </div>

            {/* SETTINGS */}

            <div className="settings">
              <div>
                <label>
                  Delay between 2 emails
                </label>

                <input
                  type="number"
                  min="0"
                  value={delay}
                  onChange={(e) => setDelay(e.target.value)}
                />
              </div>

              <div>
                <label>
                  Hourly Limit
                </label>

                <input
                  type="number"
                  min="0"
                  value={hourlyLimit}
                  onChange={(e) => setHourlyLimit(e.target.value)}
                />
              </div>
            </div>

            {/* EDITOR */}

            <div className="editor">
              <div className="editor-toolbar">
                <button>↶</button>
                <button>↷</button>

                <span>|</span>

                <button>Tt</button>
                <button><b>B</b></button>
                <button><i>I</i></button>
                <button><u>U</u></button>

                <span>|</span>

                <button>≡</button>
                <button>☷</button>

                <span>|</span>

                <button>☰</button>
                <button>☷</button>

                <span>|</span>

                <button>↔</button>
                <button>❝</button>
              </div>

              <textarea
                value={body}
                onChange={(e) => setBody(e.target.value)}
                placeholder="Type Your Reply..."
              />
            </div>
          </div>

          {/* SEND LATER POPUP */}

          {showSchedule && (
            <div className="schedule-popup">
              <h3>
                Send Later
              </h3>

              <label>
                Pick date & time
              </label>

              <input
                type="datetime-local"
                value={scheduleDate}
                min={new Date(
                  Date.now() - new Date().getTimezoneOffset() * 60000
                )
                  .toISOString()
                  .slice(0, 16)}
                onChange={(e) => setScheduleDate(e.target.value)}
              />

              <div className="quick-times">
                <button
                  onClick={() => {
                    const d = new Date();
                    d.setDate(d.getDate() + 1);
                    d.setHours(10, 0, 0, 0);

                    const localDate = new Date(
                      d.getTime() - d.getTimezoneOffset() * 60000
                    )
                      .toISOString()
                      .slice(0, 16);

                    setScheduleDate(localDate);
                  }}
                >
                  Tomorrow, 10:00 AM
                </button>

                <button
                  onClick={() => {
                    const d = new Date();
                    d.setDate(d.getDate() + 1);
                    d.setHours(11, 0, 0, 0);

                    const localDate = new Date(
                      d.getTime() - d.getTimezoneOffset() * 60000
                    )
                      .toISOString()
                      .slice(0, 16);

                    setScheduleDate(localDate);
                  }}
                >
                  Tomorrow, 11:00 AM
                </button>

                <button
                  onClick={() => {
                    const d = new Date();
                    d.setDate(d.getDate() + 1);
                    d.setHours(15, 0, 0, 0);

                    const localDate = new Date(
                      d.getTime() - d.getTimezoneOffset() * 60000
                    )
                      .toISOString()
                      .slice(0, 16);

                    setScheduleDate(localDate);
                  }}
                >
                  Tomorrow, 3:00 PM
                </button>
              </div>

              <div className="schedule-footer">
                <button
                  className="cancel-btn"
                  onClick={() => setShowSchedule(false)}
                >
                  Cancel
                </button>

                <button
                  className="done-btn"
                  onClick={scheduleEmail}
                >
                  Done
                </button>
              </div>
            </div>
          )}
        </div>
      )}
    </div>
  );
}

export default App;
